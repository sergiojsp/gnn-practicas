/* mostrar grafo de actores y peliculas */ 
MATCH (p:Person)-[r:ACTED_IN]->(m:Movie) RETURN p,r,m

/* mostrar grafo completo */ 
MATCH (n)-[r]->(m) RETURN n, r, m

/* mostrar todos los nodos */
MATCH (n) RETURN n

/* mostrar 100 nodos */
MATCH (n) RETURN n LIMIT 100

/* lista de nombre y año de nacimiento de personas*/
MATCH (p:Person) RETURN p.name AS nombre, p.born as born

/* personas que actuaron en películas */
MATCH (a:Person)-[r:ACTED_IN] -> (m:Movie) RETURN a, r, m

/* películas donde actuó Keanu Reeves */
MATCH (a:Person {name: 'Keanu Reeves'})-[r:ACTED_IN] -> (m:Movie) RETURN a,r,m

/* personas que actuaron y dirigieron películas */
MATCH (a:Person)-[r:ACTED_IN] -> (m:Movie), (a)-[d:DIRECTED]-(m) RETURN a, r, d, m

/* personas que dirigieron, produjeron y escribieron películas */
MATCH (a:Person)-[r:DIRECTED] -> (m:Movie), (a)-[d:PRODUCED]-(m), (a)-[w:WROTE]-(m) RETURN a, r, d, w, m

/* personas que siguen a otros */
MATCH (a:Person)-[r:FOLLOWS]->(b:Person) RETURN a, r, b

/* actores que han trabajado juntos en la misma película */
MATCH (a:Person)-[:ACTED_IN]->(m:Movie)<-[:ACTED_IN]-(b:Person)
WHERE a.name < b.name
WITH a, b, collect(m.title) AS movies, count(m) AS moviesTogether
WHERE moviesTogether > 2
RETURN a.name, b.name, movies, moviesTogether
ORDER BY moviesTogether DESC

/* crear una relación FOLLOWS entre dos personas */
MATCH (a:Person {name: 'Jessica Thompson'})
MATCH (b:Person {name: 'James Thompson'})
MERGE (a)-[r:FOLLOWS]->(b)

/* buscar la ruta más corta entre dos actores */
MATCH (p1:Person {name: "Tom Hanks"}), 
      (p2:Person {name: "Kevin Bacon"}),
      path = shortestPath((p1)-[:ACTED_IN*]-(p2))
RETURN path

/* borrar una relación FOLLOWS entre dos personas
MATCH (a:Person {name: 'Jessica Thompson'})-[r:FOLLOWS]->(b:Person {name: 'James Thompson'})
DELETE r;

/* borrar el grafo completo */
MATCH (n) DETACH DELETE n

/*borrar constraint de la base de datos*/
DROP CONSTRAINT movie_title IF EXISTS;
DROP CONSTRAINT person_name IF EXISTS;
